def mask_fences(text: str) -> str:
    """Keep line numbers while excluding fenced examples from structural parsing."""
    output, active = [], None
    for line in text.splitlines(keepends=True):
        marker = re.match(r"^\s{0,3}(`{3,}|~{3,})", line)
        if marker:
            fence = marker.group(1)
            if active is None:
                active = (fence[0], len(fence))
            elif fence[0] == active[0] and len(fence) >= active[1]:
                active = None
            output.append("\n" if line.endswith("\n") else "")
        elif active is None:
            output.append(line)
        else:
            output.append("\n" if line.endswith("\n") else "")
    return "".join(output)


def section_spans(text: str, heading_pattern: str):
    """Find selected ATX-heading sections up to the next peer/parent heading."""
    headings = list(re.finditer(r"^(#{1,6})\s+(.+)$", mask_fences(text), re.M))
    spans = []
    for index, heading in enumerate(headings):
        if not re.search(heading_pattern, heading.group(2), re.I):
            continue
        end = len(text)
        for following in headings[index + 1 :]:
            if len(following.group(1)) <= len(heading.group(1)):
                end = following.start()
                break
        spans.append((heading.start(), end))
    return spans


def pick_sections(text: str, pattern: str) -> str:
    return "\n".join(text[start:end] for start, end in section_spans(text, pattern))


def omit_sections(text: str, pattern: str) -> str:
    for start, end in reversed(section_spans(text, pattern)):
        text = text[:start] + text[end:]
    return text


def normalized_spacing(text: str) -> str:
    # Only allow blank separator spacing around inserted module sections.
    return re.sub(r"\n{3,}", "\n\n", text.replace("\r\n", "\n")).strip()


def table_cells(line: str) -> list[str]:
    """Split GFM pipes; an odd number of preceding slashes escapes the pipe."""
    cells, current = [], []
    for char in line.strip():
        if char == "|":
            backslashes = 0
            for previous in reversed(current):
                if previous != "\\":
                    break
                backslashes += 1
            if backslashes % 2 == 0:
                cells.append("".join(current).strip())
                current = []
                continue
        current.append(char)
    cells.append("".join(current).strip())
    if cells and cells[0] == "":
        cells = cells[1:]
    if cells and cells[-1] == "":
        cells = cells[:-1]
    return cells


def own_text(path: str, text: str) -> str:
    if path in MODULE_FILES:
        return pick_sections(text, r"^orchestration\b")
    if path.endswith("invariants.md"):
        return pick_sections(text, r"^INV-(?:003|006|007)\b")
    if path.endswith("evaluation-table.md"):
        return pick_sections(text, r"^EVAL-002\b")
    if path.endswith("trace-matrix.md"):
        return "\n".join(line for line in text.splitlines() if re.match(r"^\| REQ-00[13] \|", line))
    if path.endswith("review-memo.md"):
        return pick_sections(text, r"^PREP-CENTAY-20261114\b")
    return text


def expand_refs(text: str) -> set[str]:
    refs = set(re.findall(r"\b(?:INV|POL|EVAL|ADR)-\d{3}\b", text))
    for match in re.finditer(r"\b(INV|POL|EVAL|ADR)-(\d{3})(?:/(\d{3}))+", text):
        prefix = match.group(1)
        refs.update(f"{prefix}-{number}" for number in re.findall(r"\d{3}", match.group(0)))
    for match in re.finditer(r"\b(INV|POL|EVAL|ADR)-(\d{3})\s*[–—-]\s*(?:(INV|POL|EVAL|ADR)-)?(\d{3})\b", text):
        prefix, start, other_prefix, end = match.groups()
        if other_prefix not in (None, prefix):
            continue
        if 0 <= int(end) - int(start) <= 100:
            refs.update(f"{prefix}-{number:03d}" for number in range(int(start), int(end) + 1))
    return refs


def load_yaml_module():
    try:
        return importlib.import_module("yaml"), "installed Python module"
    except ModuleNotFoundError:
        archive = Path.home() / ".cache/uv/archive-v0"
        candidates = list(archive.glob("*/yaml/__init__.py"))
        candidates += list(archive.glob("*/lib/python*/site-packages/yaml/__init__.py"))
        for candidate in sorted(candidates):
            module_root = candidate.parent.parent
            sys.path.insert(0, str(module_root))
            try:
                return importlib.import_module("yaml"), str(module_root)
            except (ModuleNotFoundError, ImportError):
                sys.path.remove(str(module_root))
        raise RuntimeError("PyYAML is unavailable; no package installation was attempted")

    UniqueKeyLoader.add_constructor(yaml_mod.resolver.BaseResolver.DEFAULT_MAPPING_TAG, unique_mapping)
    def read_yaml(text):
        return yaml_mod.load(text, Loader=UniqueKeyLoader)
    try:
        ai_data = read_yaml(current("docs/ai-use-log.yaml"))

    ]

        "tool_kind": "external static SA document audit, not product tests",
        "generated_at": datetime.now(ZoneInfo("Asia/Taipei")).isoformat(),
        "invocation": [sys.executable, str(Path(__file__).resolve()), *sys.argv[1:]],
        "tool_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "repository": str(repo), "baseline": args.baseline,
        "static_verdict": static_verdict, "failed_checks": failure_names, "pending_checks": pending_names,
        "checks": checks, "commands": commands, "gated_followups": followups,
        "limitations": ["Static syntax/scope/traceability checks do not prove design correctness or runtime fit.", "External proposed ADR references resolve source provenance only; no shared-reader approval is inferred.", "No secrets, project .env files, direct .git files, network request or model API is read/run by this tool."],
    }
    prefix.parent.mkdir(parents=True, exist_ok=True)
    prefix.with_suffix(".json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    markdown = ["# Static SA document validation", "", f"Generated: {result['generated_at']}", f"Static verdict: **{static_verdict}**", "", "This is an external document audit. No functional, regression, live-test or human-approval PASS is implied.", "", "## Command", "", "```text", " ".join(result["invocation"]), "```", "", "## Checks", "", "| Static check | Verdict |", "|---|---|"]
    markdown += [f"| {item['name']} | {item['verdict']} |" for item in checks]
    markdown += ["", "## Findings", ""]
    for item in checks:
        if item["verdict"] != "PASS":
            markdown += [f"### {item['name']} — {item['verdict']}", "", "```json", json.dumps(item["details"], ensure_ascii=False, indent=2), "```", ""]
    if not failure_names and not pending_names:
        markdown += ["No static scope, structure, reference or evidence-file error was found by these checks.", ""]
    markdown += ["## Later gates and tests", "", "| Check | Verdict | Reason |", "|---|---|---|"]
    markdown += [f"| {item['check']} | {item['verdict']} | {item['reason']} |" for item in followups]
    markdown += ["", "The JSON report contains exact Git commands, exit codes, outputs, the changed-file inventory and detailed check results.", ""]
    prefix.with_suffix(".md").write_text("\n".join(markdown))
    print(json.dumps({"static_verdict": static_verdict, "failed_checks": failure_names, "pending_checks": pending_names, "checks": len(checks), "json_report": str(prefix.with_suffix('.json')), "markdown_report": str(prefix.with_suffix('.md'))}, ensure_ascii=False, indent=2))
    return 1 if failure_names else 0
